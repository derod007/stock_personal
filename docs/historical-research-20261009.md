```powershell
php bin/paper_historical_research.php --account=paper-kr --source-dir=docs/paper-kr-5d-source --as-of=2026-10-01T04:02:43+09:00 > historical-current.json
```

```powershell
$previousPaperState = $env:PAPER_STATE_DIR
$env:PAPER_STATE_DIR = Join-Path $env:TEMP ('stock-history-' + [guid]::NewGuid())
```
